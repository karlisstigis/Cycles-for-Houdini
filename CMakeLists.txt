# Community builds (docs/dev/community-builds.md) give DLSS an NVIDIA project id of their own
# (build-identity.json, passed by scripts/build.ps1 -DlssProjectId). Empty, as on the official
# route: nothing here happens and the engine source is compiled as it is. Otherwise a copy of
# denoiser_dlss.cpp in which only the ngx_project_id text differs is compiled in its place;
# third_party/cycles, patches/ and source.lock.json stay as they are.
set(CH_DLSS_PROJECT_ID "" CACHE STRING "NVIDIA NGX project id for DLSS (empty: the engine's own)")
set(CH_DLSS_COPY_DIR "${CMAKE_BINARY_DIR}/community-dlss")
if(CH_DLSS AND CH_DLSS_PROJECT_ID)
  string(REPEAT "[0-9a-f]" 4 ch_hex4) # CMake regular expressions have no {n}
  if(NOT CH_DLSS_PROJECT_ID MATCHES "^${ch_hex4}${ch_hex4}-${ch_hex4}-${ch_hex4}-${ch_hex4}-${ch_hex4}${ch_hex4}${ch_hex4}$")
    message(FATAL_ERROR "CH_DLSS_PROJECT_ID must be a lowercase GUID such as a0f57b54-1daf-4934-90ae-c4035c19df04")
  endif()
  set(ch_dlss_source "${CH_ENGINE_ROOT}/src/integrator/denoiser_dlss.cpp")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ch_dlss_source}")
  file(READ "${ch_dlss_source}" ch_dlss_text)
  # No ";" in the pattern: CMake would read it as a list separator.
  set(ch_dlss_pattern "static constexpr char ngx_project_id\\[\\] = \"[0-9a-f-]+\"")
  string(REGEX MATCHALL "${ch_dlss_pattern}" ch_dlss_found "${ch_dlss_text}")
  list(LENGTH ch_dlss_found ch_dlss_count)
  if(NOT ch_dlss_count EQUAL 1)
    message(FATAL_ERROR "Expected one ngx_project_id in ${ch_dlss_source}, found ${ch_dlss_count}")
  endif()
  string(REGEX REPLACE "${ch_dlss_pattern}" "static constexpr char ngx_project_id[] = \"${CH_DLSS_PROJECT_ID}\""
    ch_dlss_text "${ch_dlss_text}")
  set(ch_dlss_copy "${CH_DLSS_COPY_DIR}/denoiser_dlss.cpp")
  set(ch_dlss_old "")
  if(EXISTS "${ch_dlss_copy}")
    file(READ "${ch_dlss_copy}" ch_dlss_old)
  endif()
  if(NOT ch_dlss_old STREQUAL ch_dlss_text)
    file(WRITE "${ch_dlss_copy}" "${ch_dlss_text}") # rewritten only when it changes
  endif()
  get_target_property(ch_dlss_sources cycles_integrator SOURCES)
  list(FIND ch_dlss_sources "denoiser_dlss.cpp" ch_dlss_index)
  if(ch_dlss_index EQUAL -1)
    list(FIND ch_dlss_sources "${ch_dlss_source}" ch_dlss_index)
  endif()
  if(ch_dlss_index EQUAL -1)
    message(FATAL_ERROR "cycles_integrator does not list denoiser_dlss.cpp")
  endif()
  list(REMOVE_AT ch_dlss_sources ${ch_dlss_index})
  list(INSERT ch_dlss_sources ${ch_dlss_index} "${ch_dlss_copy}")
  set_target_properties(cycles_integrator PROPERTIES SOURCES "${ch_dlss_sources}")
  if(MSVC)
    set_property(SOURCE "${ch_dlss_copy}" DIRECTORY "${CH_ENGINE_ROOT}/src/integrator"
      APPEND PROPERTY COMPILE_OPTIONS /Zc:preprocessor)
  endif()
  message(STATUS "DLSS: community project id ${CH_DLSS_PROJECT_ID} (compiled from ${ch_dlss_copy})")
elseif(EXISTS "${CH_DLSS_COPY_DIR}")
  file(REMOVE_RECURSE "${CH_DLSS_COPY_DIR}")
endif()
